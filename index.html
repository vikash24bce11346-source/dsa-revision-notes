Two Pointers

Used in sorted arrays and pair finding problems. 


void twoPointers(vector& arr) {

    int l = 0;
    int r = arr.size() - 1;

    while (l < r) {

        if (condition_met(arr[l], arr[r])) {
            l++;
            r--;
        }
        else if (need_larger) {
            l++;
        }
        else {
            r--;
        }
    }
}

Sliding Window

Used in substring and subarray problems. 


int slidingWindow(string s) {

    int l = 0;
    int best = 0;

    unordered_map freq;

    for (int r = 0; r < s.size(); r++) {

        freq[s[r]]++;

        while (window_invalid(freq)) {

            freq[s[l]]--;

            if (freq[s[l]] == 0)
                freq.erase(s[l]);

            l++;
        }

        best = max(best, r - l + 1);
    }

    return best;
}
